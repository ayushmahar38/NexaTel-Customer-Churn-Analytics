import os

# Create a output directory for cleaned files
output_dir = "./cleaned_data"
os.makedirs(output_dir, exist_ok=True)

# 1. Clean Billing Data (remove negative total amounts)
df_billing = dataframes['billing'].copy()
df_billing_clean = df_billing[df_billing['total_amount_inr'] >= 0]
print(f"Billing: Removed {len(df_billing) - len(df_billing_clean)} negative rows.")

# 2. Clean Payments Data (remove records with broken or missing invoice IDs)
df_payments = dataframes['payments'].copy()
valid_invoices = set(df_billing_clean['invoice_id'].dropna())
df_payments_clean = df_payments[df_payments['invoice_id'].isin(valid_invoices)]
print(f"Payments: Removed {len(df_payments) - len(df_payments_clean)} invalid/orphaned payment records.")

# 3. Save all cleaned dataframes into the cleaned_data folder
cleaned_dict = dataframes.copy()
cleaned_dict['billing'] = df_billing_clean
cleaned_dict['payments'] = df_payments_clean

for name, df in cleaned_dict.items():
    file_path = os.path.join(output_dir, f"{name}_cleaned.csv")
    df.to_csv(file_path, index=False)

print(f"\n All 24 dataset files cleaned and saved to '{output_dir}/'")

sql_content = """-- NexaTel Customer Churn Analytics - SQL Analysis Queries

-- 1. Demographics and Churn Distribution
SELECT 
    gender,
    senior_citizen,
    COUNT(customer_id) AS total_customers,
    SUM(CASE WHEN churn_date IS NOT NULL THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(AVG(CASE WHEN churn_date IS NOT NULL THEN 1.0 ELSE 0.0 END) * 100, 2) AS churn_rate_pct
FROM customers_cleaned
GROUP BY gender, senior_citizen
ORDER BY churn_rate_pct DESC;

-- 2. Churn Rate by Contract Type & Product Line
SELECT 
    con.contract_type,
    COUNT(c.customer_id) AS total_accounts,
    SUM(CASE WHEN c.churn_date IS NOT NULL THEN 1 ELSE 0 END) AS churned_accounts,
    ROUND(AVG(CASE WHEN c.churn_date IS NOT NULL THEN 1.0 ELSE 0.0 END) * 100, 2) AS churn_rate_pct
FROM customers_cleaned c
JOIN contracts_cleaned con ON c.customer_id = con.customer_id
GROUP BY con.contract_type
ORDER BY churn_rate_pct DESC;

-- 3. Churn Rate by Complaint Status
SELECT 
    comp.status AS complaint_status,
    COUNT(DISTINCT c.customer_id) AS impacted_customers,
    SUM(CASE WHEN c.churn_date IS NOT NULL THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(AVG(CASE WHEN c.churn_date IS NOT NULL THEN 1.0 ELSE 0.0 END) * 100, 2) AS churn_rate_pct
FROM customers_cleaned c
LEFT JOIN complaints_cleaned comp ON c.customer_id = comp.customer_id
WHERE comp.status IS NOT NULL
GROUP BY comp.status;

-- 4. Total Revenue Impact of Churned Accounts
SELECT 
    c.churn_status,
    COUNT(DISTINCT c.customer_id) AS customer_count,
    ROUND(SUM(b.total_amount_inr), 2) AS total_revenue_inr,
    ROUND(AVG(b.total_amount_inr), 2) AS avg_revenue_per_customer_inr
FROM customers_cleaned c
JOIN billing_cleaned b ON c.customer_id = b.customer_id
GROUP BY c.churn_status;
"""

with open("./cleaned_data/NexaTel_Churn_Queries.sql", "w") as f:
    f.write(sql_content)
